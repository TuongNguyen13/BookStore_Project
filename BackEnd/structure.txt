│   │   ├── css/

│   └── README.md

│   ├── config/       # Cấu hình cơ sở dữ liệu, biến môi trường

│   │   └── product.model.js
│   ├── routes/       # Định nghĩa các API endpoint

│   │   └── productRoutes.js
│   ├── services/     # Logic nghiệp vụ, thường gọi các hàm từ models
│   │   ├── authService.js
│   │   └── productService.js
│   ├── utils/        # Các hàm tiện ích chung
│   │   └── helpers.js
│   ├── app.js        # File khởi tạo server chính
│   ├── package.json
│   ├── .env          # Biến môi trường (port, chuỗi kết nối DB, v.v.)
